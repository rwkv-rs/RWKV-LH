"""Frozen-source download and audit campaign; no model execution or training."""
import concurrent.futures
import fcntl
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

ROOT=Path('/home/chase/GitHub/RWKV-LH')
P=ROOT/'data/experiments/RWKV_SFT_BACKGROUND_R8_20260915'
AUDIT_ROOT=P/'audit_runtime'
REV='f684cc1a9f3e19f6f4929102cd9b06cc0b895b8a'

def save(path,value):
    tmp=path.with_suffix(path.suffix+'.tmp')
    with tmp.open('w') as f:
        json.dump(value,f,ensure_ascii=False,indent=2);f.flush();os.fsync(f.fileno())
    os.replace(tmp,path)

def digest(path):
    with path.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()

def verify_file(path,item):
    if path.stat().st_size!=item['size'] or digest(path)!=item['lfs']['oid']:
        raise ValueError('full shard size/SHA differs from frozen source; not admitted to audit')

def work(item):
    name=Path(item['path']).stem;folder=P/name;folder.mkdir(exist_ok=True)
    state={'shard':name,'revision':REV,'source_path':item['path'],'expected_bytes':item['size'],'expected_sha256':item['lfs']['oid'],'executed_tasks':0,'training_rows':0,'started_at':time.time()}
    def status(phase,**kw):state.update(phase=phase,updated_at=time.time(),**kw);save(folder/'STATUS.json',state)
    try:
        raw=folder/'source.jsonl';partial=folder/'source.jsonl.download'
        if not raw.exists():
            status('downloading')
            url='https://huggingface.co/datasets/openbmb/UltraData-SFT-Agent-2609/resolve/'+REV+'/'+item['path']+'?download=true'
            with (folder/'download.log').open('a') as log:
                subprocess.run(['curl','--fail','--location','--show-error','--silent','--retry','3','--retry-delay','5','--connect-timeout','30','--continue-at','-','--output',str(partial),'--',url],stdout=log,stderr=subprocess.STDOUT,check=True)
            status('verifying_source');verify_file(partial,item);os.replace(partial,raw)
        else:verify_file(raw,item)
        status('auditing_source_structure')
        out=folder/'audit.jsonl'
        if not out.exists():
            cmd=[str(ROOT/'.venv/bin/python'),str(AUDIT_ROOT/'scripts/audit_collection_sources.py'),'--source',str(raw),'--revision',REV,'--output',str(out)]
            if Path(str(out)+'.checkpoint.json').exists():cmd.append('--resume')
            with (folder/'audit.log').open('a') as log:subprocess.run(cmd,stdout=log,stderr=subprocess.STDOUT,check=True,cwd=ROOT)
        cp=json.loads(Path(str(out)+'.checkpoint.json').read_text())
        assert cp['prefix_sha256']==item['lfs']['oid'] and cp['offset']==item['size']
        assert cp['output_sha256']==digest(out)
        status('source_audit_complete_requires_environment',rows=cp['count'],invalid_rows=cp['invalid_rows'],audit_sha256=cp['output_sha256'],finished_at=time.time())
    except Exception as exc:
        status('failed',error={'type':type(exc).__name__,'message':str(exc)},finished_at=time.time())
    return state

def main():
    reg=json.loads((P/'REGISTRATION.json').read_text())
    for rel,sha in reg['frozen_scripts'].items():
        if digest(ROOT/rel)!=sha:raise ValueError('frozen audit source changed: '+rel)
    with (P/'campaign.lock').open('a') as lock:
        fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB)
        save(P/'STATUS.json',{'phase':'running_source_preparation','pid':os.getpid(),'started_at':time.time(),'workers':4,'executed_tasks':0,'training_rows':0})
        with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
            results=list(pool.map(work,reg['shards']))
        total=sum(r.get('rows',0) for r in results);failed=[r['shard'] for r in results if r['phase']=='failed']
        save(P/'STATUS.json',{'phase':'failed' if failed else 'source_audit_complete_requires_environment','finished_at':time.time(),'source_rows':total,'failed_shards':failed,'executed_tasks':0,'training_rows':0,'formal_admitted_tasks':0,'next':'Reconstruct and validate source environments and native conversion; do not submit static source dialogues to RWKV as production traces.'})
        return int(bool(failed))

if __name__=='__main__':raise SystemExit(main())

def write(path,value):path.parent.mkdir(parents=True,exist_ok=True);path.write_text(json.dumps(value,indent=2)+'\n')
def event(ident,kind,payload):return {'event_id':ident,'event_type':kind,'payload':payload}
def database(case,main,wal=None,shm=None):
    path=case/'state/long_horizon.db';path.parent.mkdir(parents=True,exist_ok=True)
    for suffix,size in (('',main),('-wal',wal),('-shm',shm)):
        target=Path(str(path)+suffix)
        if size is not None:
            with target.open('wb') as handle:handle.truncate(size)

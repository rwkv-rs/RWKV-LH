"""Read-only trace/history audit; writes derived records, never restores old data."""
import hashlib
import json
import subprocess
from collections import Counter
from pathlib import Path

ROOT = Path('/home/chase/GitHub/RWKV-LH')
OUT = ROOT / 'data/experiments/ARCHITECTURE_REVIEW_R1_20260921'
OUT.mkdir(parents=True, exist_ok=True)

def sha(b):
    return hashlib.sha256(b).hexdigest()

def save(name, obj):
    (OUT / name).write_text(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')

def command(*args):
    return subprocess.check_output(args, cwd=ROOT).decode()

roots = [ROOT / 'data/experiments/R38/agent-run/execution',
         ROOT / 'data/experiments/R39/advised-run/execution']
for parent in ('data/pipeline/sources', 'data/pipeline/r40/sources'):
    roots.extend(sorted(p.parent for p in (ROOT / parent).glob('*/model_trace.jsonl')))
summaries = []
evidence = {}
for root in roots:
    files = {n: root / n for n in ('model_trace.jsonl', 'state_snapshot.json', 'RESULT.json')}
    for p in files.values():
        evidence[str(p.relative_to(ROOT))] = sha(p.read_bytes())
    events = [json.loads(line) for line in files['model_trace.jsonl'].read_text().splitlines() if line]
    state = json.loads(files['state_snapshot.json'].read_text())
    result = json.loads(files['RESULT.json'].read_text())
    actions = list(state.get('actions', {}).values())
    outputs = []
    for e in events:
        if e.get('type') == 'model_session_generation_returned':
            outputs.append({k: v for k, v in e.items() if not isinstance(v, (list, dict))})
    checkpoints = list(state.get('model_states', {}).values())
    transcripts = [c.get('transcript', '') for c in checkpoints]
    summaries.append({
        'root': str(root.relative_to(ROOT)),
        'result': {k: v for k, v in result.items() if k in (
            'id', 'status', 'termination', 'termination_reason', 'final', 'generation_started',
            'protocol_rejections', 'trace_complete', 'tool_scope', 'assistance')},
        'event_counts': dict(Counter(e.get('type') for e in events)),
        'actions': [{k: a.get(k) for k in ('action_id', 'action_type', 'arguments', 'status',
            'workspace_digest_before', 'workspace_digest_after')} | {
                'success': (a.get('result') or {}).get('success'),
                'result_excerpt': json.dumps(a.get('result'), ensure_ascii=False)[:1400],
                'result_keys': list((a.get('result') or {}).keys())} for a in actions],
        'mutation_actions_by_workspace_digest': sum(bool(a.get('workspace_digest_before')) and
            bool(a.get('workspace_digest_after')) and a['workspace_digest_before'] !=
            a['workspace_digest_after'] for a in actions),
        'returned_outputs': outputs,
        'checkpoint_keys': list(checkpoints[0]) if checkpoints else [],
        'first_transcript_prefix': transcripts[0][:1000] if transcripts else '',
        'last_transcript_suffix': transcripts[-1][-2500:] if transcripts else '',
        'causal_event_counts': dict(Counter(e.get('event_type') for e in state.get('causal_records', {}).values())),
        'scope_note': 'Historical collection, not a new benchmark. Assisted state actions include parent actions.'})
save('TRACE_AUDIT.json', summaries)
save('SOURCE_SHA256.json', evidence)
baseline = {
    'head': command('git', 'rev-parse', 'HEAD').strip(),
    'status': command('git', 'status', '--short'),
    'tracked_dirty_sha256': {p: sha((ROOT / p).read_bytes()) for p in
        command('git', 'diff', '--name-only').splitlines() if (ROOT / p).is_file()},
    'analysis_only': True, 'new_agent_runs': 0, 'new_training_steps': 0}
if not (OUT / 'WORKTREE_AT_START.json').exists():
    save('WORKTREE_AT_START.json', baseline)
save('SOURCE_SCOPE.json', {'trace_roots': [str(p.relative_to(ROOT)) for p in roots],
    'protected_content_read': False,
    'protected_paths': ['data/acceptance', 'data/datasets/rwkv_lh_real_agent_holdout_v2',
        'benchmarks/rwkv_e2e/rwkv_real_agent_holdout_v2', 'acceptance_tests'],
    'scope': 'All 11 model traces in the explicit current source roots; historical reports separately indexed. Not every deleted historical raw trace.'})
for s in summaries:
    print(json.dumps({'root': s['root'], 'result': s['result'],
        'actions': [a['action_type'] for a in s['actions']],
        'mutations': s['mutation_actions_by_workspace_digest'],
        'types': s['event_counts'], 'causal': s['causal_event_counts']}, ensure_ascii=False))

import itertools,math,collections
count=0
for k in range(2,6):
 def compositions(n,p):
  if p==1:yield(n,);return
  for x in range(n+1):
   for rest in compositions(n-x,p-1):yield(x,)+rest
 for n in range(0,13):
  for start in compositions(n,k):
   if k>=5 and n>8:break
   states={start};queue=[start]
   for a in queue:
    for i in range(k):
     if all(a[j]>0 for j in range(k) if j!=i):
      b=tuple(a[j]+(k-1 if j==i else -1) for j in range(k))
      if b not in states:states.add(b);queue.append(b)
   a=start;forced={a}
   while a.count(0)==1:
    i=a.index(0);a=tuple(a[j]+(k-1 if j==i else -1) for j in range(k))
    if a in forced:break
    forced.add(a)
   if 0 not in a:
    expected=0
    for phase in range(k):
     residual=sum((x-phase)%k for x in start);q=(n-residual)//k
     if q>=0:expected+=math.comb(q+k-1,k-1)
   else:expected=len(forced)
   assert len(states)==expected,(start,len(states),expected)
   count+=1
print('hypothesis exhaustive passed',count)

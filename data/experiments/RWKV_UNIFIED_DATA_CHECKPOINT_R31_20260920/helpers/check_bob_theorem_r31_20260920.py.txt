import itertools
for n in range(1,9):
 best={}
 for p in itertools.permutations(range(n)):
  a=[];b=[0]*n
  for i in range(n):a.append(1+max((a[j] for j in range(i) if p[j]<p[i]),default=0))
  for i in range(n-1,-1,-1):b[i]=1+max((b[j] for j in range(i+1,n) if p[j]<p[i]),default=0)
  a=tuple(a);best[a]=max(best.get(a,0),sum(b))
 for a,v in best.items():
  b=[0]*n
  for i in range(n-1,-1,-1):b[i]=1+max((b[j] for j in range(i+1,n) if a[j]<=a[i]),default=0)
  assert sum(b)==v,(a,v,b)
 print(n,len(best),flush=True)

def oracle(mana,ops):
 spells=[(0,0)];out=[];last=0
 for index,(kind,a,b) in enumerate(ops,1):
  x=(a+last)%1000000+1;y=(b+last)%1000000+1
  if kind==1:spells.append((y,x));continue
  t=x;h=y;best=Fraction(0)
  for cost,damage in spells:
   if cost:best=max(best,damage*min(Fraction(mana,cost),t))
  for (ca,da),(cb,db) in itertools.combinations(spells,2):
   if ca==cb:continue

 return out

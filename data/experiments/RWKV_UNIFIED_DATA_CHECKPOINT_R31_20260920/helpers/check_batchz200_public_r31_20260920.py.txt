def brute(grid,K):
 H=len(grid);W=len(grid[0]);N=H*W;answer=0
 for bits in itertools.product(range(2),repeat=N):
  if abs(2*sum(bits)-N)>K or not 0<sum(bits)<N:continue
  if any((grid[r][q]=='T' and not bits[r*W+q]) or (grid[r][q]=='D' and bits[r*W+q]) for r in range(H) for q in range(W)):continue
  if any(len({bits[r*W+q],bits[r*W+q+1],bits[(r+1)*W+q],bits[(r+1)*W+q+1]})==1 for r in range(H-1) for q in range(W-1)):continue
  okay=True
  for colour in (0,1):
   nodes={i for i,b in enumerate(bits) if b==colour};seen={next(iter(nodes))};todo=list(seen)
   for a in todo:
    r,q=divmod(a,W)
    for rr,qq in ((r-1,q),(r+1,q),(r,q-1),(r,q+1)):
     b=rr*W+qq
     if 0<=rr<H and 0<=qq<W and b in nodes and b not in seen:seen.add(b);todo.append(b)
   if seen!=nodes:okay=False;break
  answer+=okay
 return answer

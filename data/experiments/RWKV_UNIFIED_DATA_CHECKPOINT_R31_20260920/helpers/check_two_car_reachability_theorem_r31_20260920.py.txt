import itertools,json
from pathlib import Path
W=5;positions=list(itertools.product(range(2),range(W)));states=[(a,b) for a in positions for b in positions if a!=b]
def formula(start,end,t):
 a,b=start;c,d=end
 if c==d:return False
 da=c[1]-a[1];db=d[1]-b[1]
 if min(da,db)<0 or da+(a[0]!=c[0])>t or db+(b[0]!=d[0])>t:return False
 if a[0]==b[0]==c[0]==d[0] and (a[1]-b[1])*(c[1]-d[1])<0:return t-da>=2 or t-db>=2
 return True
checks=0
for start in states:
 reached={start}
 for t in range(7):
  for end in states:
   if formula(start,end,t)!=(end in reached):print('COUNTEREXAMPLE',start,end,t,formula(start,end,t));raise SystemExit(1)
   checks+=1
  next_states=set()
  for a,b in reached:
   aa=[a,(1-a[0],a[1])]+([(a[0],a[1]+1)] if a[1]+1<W else [])
   bb=[b,(1-b[0],b[1])]+([(b[0],b[1]+1)] if b[1]+1<W else [])
   next_states.update((x,y) for x in aa for y in bb if x!=y)
  reached=next_states

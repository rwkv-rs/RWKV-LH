from itertools import product
for n in range(2,16):
 found=[set() for _ in range(n+1)]
 for s in product(range(2),repeat=n):
  a=sum(s)
  if a in (0,n):continue
  k=next(k for k in range(1,n+1) if all(s[i]==s[i%k] for i in range(n)));found[a].add(k)
 for a in range(1,n):
  b=n-a;pred=set()
  for k in range(2,n+1):
   q=n//k;al=(a+q)//(q+1);bl=(b+q)//(q+1)
   if al<=a//q and bl<=b//q and al+bl<=k<=a//q+b//q:pred.add(k)
  assert pred==found[a],(a,b,pred,found[a])
print('All binary strings lengths2..15 passed')

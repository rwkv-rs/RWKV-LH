import heapq,bisect,time
bound=[0,1];values=[0,1];events=[2,3,4,5,6];queued=set(events);start=time.monotonic();count=0
while events:
 x=heapq.heappop(events);queued.remove(x)
 if x>10**18:break
 seen={values[bisect.bisect_right(bound,x//d)-1] for d in range(2,7)};g=0
 while g in seen:g+=1
 if g!=values[-1]:
  bound.append(x);values.append(g)
  for d in range(2,7):
   v=x*d
   if v<=10**18 and v not in queued:heapq.heappush(events,v);queued.add(v)
 count+=1
 if count>2000000:raise RuntimeError('event budget exceeded')
print(len(bound),count,time.monotonic()-start)
print(list(zip(bound[:30],values[:30])))

def round_outcomes(teams):

 for choices in itertools.product((0,1),repeat=len(pairs)):

def histories(n):
 answer=[]
 def play(upper,lower,games):
  if len(upper)==1:answer.append(games+((1<<upper[0])|(1<<lower[0]),));return

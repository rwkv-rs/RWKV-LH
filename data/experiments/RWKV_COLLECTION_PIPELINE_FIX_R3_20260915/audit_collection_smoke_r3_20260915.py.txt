from pathlib import Path
import sys,json,hashlib
root=Path('/home/chase/GitHub/RWKV-LH');sys.path.insert(0,str(root))
from rwkv_lh.schema import RunState
from rwkv_lh.model_io import render_bootstrap,render_event_append
from rwkv_lh.model import LongHorizonModel
from rwkv_lh.model_session import ModelSession
from rwkv_lh.runtime.settings import RuntimeSettings
from rwkv_lh.harness import ActionHarness
from rwkv_lh.token_budget import tokenizer
from rwkv_lh.correction_snapshots import validate_generation_snapshot
base=root/'data/experiments/RWKV_COLLECTION_PIPELINE_FIX_R3_20260915'
rows=[]
for directory in sorted((base/'smoke_runs').iterdir()):
 p=directory/'execution'
 state=RunState.from_dict(json.loads((p/'state_snapshot.json').read_text()))
 events=[json.loads(line) for line in (p/'model_trace.jsonl').read_text().splitlines()]
 starts={e['request_id']:e for e in events if e['type']=='model_session_generation_started'}
 generated={e['candidate_checkpoint_id']:e['raw_generation'] for e in events if e['type']=='model_session_generation_returned'}
 source_files={str(f.relative_to(p)):hashlib.sha256(f.read_bytes()).hexdigest() for f in p.rglob('*') if f.is_file()}
 model=LongHorizonModel(ModelSession(client=object(),settings=RuntimeSettings(base_url='http://unused.invalid',api_key='',model='audit',tool_disclosure_mode='full')),harness=ActionHarness())
 initial=RunState(run_id=state.run_id,goal=state.goal)
 bootstrap=render_bootstrap(model.direct_definitions(),model._assignment(initial,recent_limit=None))
 calls=[]
 for event in events:
  if event['type']!='model_session_generation_returned':continue
  start=starts[event['request_id']];raw=event['raw_generation']
  validate_generation_snapshot(p,start,source_files)
  cp=state.model_states[start['input_checkpoint_id']];chain=[];seen=set()
  while cp:
   assert cp.checkpoint_id not in seen;seen.add(cp.checkpoint_id);chain.append(cp)
   cp=state.model_states.get(cp.parent_checkpoint_id)
  assert chain[-1].transcript==bootstrap
  ids=[];observed=set()
  for cp in reversed(chain):
   assert cp.state_profile_id=='zero' and cp.state_profile_sha256=='0'*64
   if cp.parent_checkpoint_id:
    parent=state.model_states[cp.parent_checkpoint_id]
    assert cp.native_state_metadata['cache_binding']['parent_state_digest']==parent.native_state_digest
   if cp.checkpoint_id in generated:ids.extend(generated[cp.checkpoint_id]['raw_token_ids'])
   else:
    if cp.parent_checkpoint_id:
     delta=set(cp.event_ids)-set(parent.event_ids);assert len(delta)==1
     eid=next(iter(delta));assert eid not in observed;observed.add(eid)
     assert render_event_append(state.model_events[eid],previous_transcript=parent.transcript)==cp.transcript
    ids.extend(tokenizer().encode(cp.transcript))
  assert raw['prompt_token_ids']==[0]+ids
  assert start['input_digest']==state.model_states[start['input_checkpoint_id']].native_state_digest
  inputs=p/'actual_inputs';inputs.mkdir(exist_ok=True)
  text=tokenizer().decode(ids);(inputs/(event['request_id']+'.txt')).write_text(text)
  calls.append({'request_id':event['request_id'],'input_tokens':len(raw['prompt_token_ids']),
    'output_tokens':len(raw['raw_token_ids']),'input_sha256':hashlib.sha256(text.encode()).hexdigest(),
    'events':sorted(observed),'snapshot_valid':True})
 result=json.loads((directory/'COLLECTION_RECEIPT.json').read_text())['result']
 for action in result['actions']:
  assert 'EV-ACTION-'+action['action_id'] in calls[-1]['events']
 rows.append({'id':directory.name,'calls':calls,'trace_complete':result['trace_complete'],
              'changed_files':result['changed_files'],'final':result['final'],
              'termination':result['termination'],'protocol_rejections':result['protocol_rejections'],
              'elapsed_seconds':result['elapsed_seconds'],'mechanical_audit':True})
(base/'SMOKE_AUDIT.json').write_text(json.dumps(rows,ensure_ascii=False,indent=2))
print(json.dumps({'tasks':len(rows),'generations':sum(len(r['calls']) for r in rows),
 'input_tokens':sum(c['input_tokens'] for r in rows for c in r['calls']),
 'output_tokens':sum(c['output_tokens'] for r in rows for c in r['calls']),
 'elapsed_seconds_sum':sum(r['elapsed_seconds'] for r in rows)},indent=2))

import concurrent.futures
import hashlib
import json
from pathlib import Path
import urllib.request

ROOT=Path('/home/chase/GitHub/RWKV-LH/data/experiments/RWKV_SFT_ENVIRONMENT_R5_20260915')
REV='f684cc1a9f3e19f6f4929102cd9b06cc0b895b8a'
LIMIT=2*1024*1024

def fetch(item):
    url='https://huggingface.co/datasets/openbmb/UltraData-SFT-Agent-2609/resolve/'+REV+'/'+item['path']
    request=urllib.request.Request(url,headers={'Range':f'bytes=0-{LIMIT-1}'})
    with urllib.request.urlopen(request,timeout=60) as response:
        raw=response.read(LIMIT)
        headers=dict(response.headers)
        status=response.status
    name=Path(item['path']).stem
    (ROOT/(name+'.prefix')).write_bytes(raw)
    lines=raw.split(b'\n')[:-1]
    rows=[json.loads(line) for line in lines if line.strip()]
    info={'path':item['path'],'revision':REV,'status':status,'headers':headers,'bytes':len(raw),
          'prefix_sha256':hashlib.sha256(raw).hexdigest(),'full_source_lfs_sha256':item['lfs']['oid'],
          'complete_rows':len(rows),'full_file_verified':False,'rows':[{'uuid':r.get('uuid'),'source':r.get('source'),
          'keys':list(r),'initial_user':next((m.get('content') for m in r['messages'] if m.get('role')=='user'),None)} for r in rows]}
    (ROOT/(name+'.json')).write_text(json.dumps(info,ensure_ascii=False,indent=2))
    print(name,len(rows),sorted(set(r.get('source','') for r in rows)),flush=True)

if __name__=='__main__':
    with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
        futures=[pool.submit(fetch,item) for item in json.loads((ROOT/'SOURCE_TREE.json').read_text())]
        for future in futures:
            try:future.result()
            except Exception as exc:print(type(exc).__name__,str(exc),flush=True)

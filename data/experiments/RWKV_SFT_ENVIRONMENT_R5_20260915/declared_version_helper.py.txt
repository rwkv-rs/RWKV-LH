import hashlib
import json
import re
from pathlib import Path

root=Path('/home/chase/GitHub/RWKV-LH')
p=root/'data/experiments/RWKV_SFT_ENVIRONMENT_R5_20260915'
source=root/'data/experiments/RWKV_ULTRADATA_EXPANSION_R1_20260915/Code-Agent_ROWS.json'
row=json.loads(source.read_text())['rows'][0]['row']
observed=row['messages'][6]['content']
reference=(p/'ANYIO_4_6_0_tls.py.txt').read_text().splitlines()
matches=[];different=[]
for line in observed.splitlines():
    match=re.fullmatch(r'\s*(\d+)\|(.*)',line)
    if not match or '...(truncated)' in line:continue
    n=int(match[1]);text=match[2]
    (matches if n<=len(reference) and text==reference[n-1] else different).append(n)
result={'source_uuid':row['uuid'],'source_message_index':6,'observation_before_first_edit':True,
        'declared_package_version':'AnyIO 4.6.0','repository':'https://github.com/agronholm/anyio',
        'public_tag_commit':json.loads((p/'ANYIO_DECLARED_VERSION_REF.json').read_text())['object']['sha'],
        'reference_path':'src/anyio/streams/tls.py','reference_sha256':hashlib.sha256((p/'ANYIO_4_6_0_tls.py.txt').read_bytes()).hexdigest(),
        'equal_line_count':len(matches),'different_line_numbers':different,
        'original_observation_truncated':'...(truncated)' in observed,
        'original_environment_verified':False,'admitted':False,
        'conclusion':'Declared release is a candidate only. Partial file comparison cannot bind whole repository, altered tests, dependencies or original task environment.'}
(p/'DECLARED_VERSION_CHECK.json').write_text(json.dumps(result,ensure_ascii=False,indent=2))
print(json.dumps(result,ensure_ascii=False))

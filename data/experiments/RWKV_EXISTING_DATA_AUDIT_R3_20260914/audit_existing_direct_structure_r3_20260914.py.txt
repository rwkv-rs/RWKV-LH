from pathlib import Path
import sys,json,hashlib
R=Path('/home/chase/GitHub/RWKV-LH');sys.path.insert(0,str(R))
from rwkv_lh import model_io
from rwkv_lh.direct_trace_data import replay_run,normalize_direct_row,executed_arguments
from rwkv_lh.schema import RunState
from rwkv_lh.task_review import write_once
D=R/'data/experiments/RWKV_EXISTING_DATA_AUDIT_R3_20260914';P=R/'data/datasets/rwkv_direct_fact_fidelity_v1'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
m=json.loads((P/'manifest.json').read_text());rows=[json.loads(s) for s in (P/'train.jsonl').read_text().splitlines()]
regpath=Path(m['freeze_registration']['path']);assert sha(regpath)==m['freeze_registration']['sha256']
reg=json.loads(regpath.read_text());checks=[];sourcechecks=[]
for source in m['sources']:
 sid=source['source_id'];selected=[r for r in rows if r['source_id']==sid]
 if not selected:continue
 source_record={'source_id':sid,'valid':False};actual={};state=None
 try:
  mp=Path(source['manifest']['path']);assert sha(mp)==source['manifest']['sha256'],'source manifest differs'
  sm=json.loads(mp.read_text());root=Path(source['run_root'])
  assert sm['source_type']=='native_production_trace' and sm['model_sha256']==m['model_sha256']
  assert sm['collector_source_manifest_sha256'] and sm['server_identity_sha256']
  for name,h in sm['files'].items():
   rel=Path(name);assert not rel.is_absolute() and '..' not in rel.parts
   assert sha(root/rel)==h,'source artifact differs: '+name
  actual=replay_run(root,m['model_sha256'])
  state=RunState.from_dict(json.loads((root/'state_snapshot.json').read_text()))
  source_record.update(valid=True,generations_replayed=len(actual),sealed_files=len(sm['files']))
 except Exception as e:source_record['error']=type(e).__name__+': '+str(e)
 sourcechecks.append(source_record)
 for row in selected:
  item={'sample_id':row['sample_id'],'source_id':sid,'authority':row['label_authority'],'valid':False}
  try:
   assert source_record['valid'],'source verification failed'
   normalize_direct_row(row,model_sha256=m['model_sha256'],context_tokens=reg['context_tokens'],
      vocab_size=reg['vocab_size'],bos_token_id=reg['bos_token_id'])
   assert row['source_manifest_sha256']==source['manifest']['sha256'] and row['family']==source['family']
   assert row['source_content_sha256']==source['source_content_sha256']
   generated=actual[row['candidate_checkpoint_id']]
   for k in ('input_token_ids','input_text','input_checkpoint_id','request_id'):assert row[k]==generated[k],k+' differs'
   target=model_io.parse_model_command(row['target_text'][:-len(model_io.JSON_CALL_STOP_SUFFIXES[0])])
   if row['label_authority']=='executed_read':
    action=state.actions[row['executed_action_id']]
    original=model_io.parse_model_command(generated['raw_generation']['raw_output'])
    assert target==original and action.action_type==target.name
    assert dict(action.arguments)==executed_arguments(target) and action.result['success'] is True
    assert target.arguments['path']==source['path']
    assert hashlib.sha256(action.result['output'].encode()).hexdigest()==row['source_content_sha256']
    item['actual_action_id']=action.action_id
   else:
    visible=set();cp=state.model_states[generated['input_checkpoint_id']]
    while cp is not None:visible.update(cp.event_ids);cp=state.model_states.get(cp.parent_checkpoint_id)
    contents=[]
    for event_id in visible:
     observation=state.model_events[event_id].payload.get('result',{}).get('observation',{})
     if observation.get('projection_complete'):
      text=''.join(span['content'] for span in observation.get('exact_spans',[]))
      contents.append(hashlib.sha256(text.encode()).hexdigest())
    assert row['source_content_sha256'] in contents,'source was not visible'
   item.update(valid=True,input_tokens=len(row['input_token_ids']),target_tokens=len(row['target_token_ids']))
  except Exception as e:item['error']=type(e).__name__+': '+str(e)
  checks.append(item)
 assert sha(P/'train.jsonl')==m['train']['sha256']
write_once(D/'STRUCTURE_AUDIT.json',dict(sources=sourcechecks,rows=checks,passed=sum(x['valid'] for x in checks),
 total=len(checks),context_tokens=reg['context_tokens'],semantic_truth_verified=False,
 note='Current production replay and existing normalizer, actual read result and pre-generation visible observation only. No source mutation, dataset publication or confirmation content reading.'))
print('structural',sum(x['valid'] for x in checks),'/',len(checks))
print('errors',[(x['sample_id'],x.get('error')) for x in checks if not x['valid']][:10])

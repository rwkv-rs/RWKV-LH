from pathlib import Path
import sys,json,hashlib,subprocess
R=Path('/home/chase/GitHub/RWKV-LH');sys.path.insert(0,str(R))
from rwkv_lh.task_review import write_once
from rwkv_lh import model_io
D=R/'data/experiments/RWKV_EXISTING_DATA_AUDIT_R3_20260914';D.mkdir(exist_ok=True)
P=R/'data/datasets/rwkv_direct_fact_fidelity_v1'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
m=json.loads((P/'manifest.json').read_text());rows=[json.loads(s) for s in (P/'train.jsonl').read_text().splitlines()]
assert len(rows)==128 and sha(P/'train.jsonl')==m['train']['sha256']
write_once(D/'PACKET_BUILD_REGISTRATION.json',dict(round=D.name,source_head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=R,text=True).strip(),
 dataset_manifest_sha256=sha(P/'manifest.json'),train_sha256=sha(P/'train.jsonl'),regression_sha256=sha(P/'regression.json'),
 script_sha256=sha(Path(__file__)),source_pins={str(p.relative_to(R)):sha(p) for p in (R/'rwkv_lh').rglob('*.py')},
 scope='Audit all 128 existing train rows: exact replay/source/protocol/State; 28 real read checks; individual manual fidelity review of 100 final targets. No confirmation content, no dataset publication or training.',
 semantic_gate='Retain if original goal main meaning and important limits are covered, every specific assertion has visible support, numeric identities faithful, requirements not implementation, proposed/previous tests not current tool execution. Ordinary details may be omitted. Ambiguity is quarantined. Preserve raw labels; do not rewrite rejected answers.',
 budget={'new_rwkv_calls':0,'new_strong_api_calls':0,'training_steps':0},expected_labels=None))
sources={s['source_id']:s for s in m['sources']}
for sid in sorted(sources):
 s=sources[sid];selected=[r for r in rows if r['source_id']==sid]
 if not selected:continue
 root=Path(s['run_root']);state=json.loads((root/'state_snapshot.json').read_text())
 full=Path(s['content_reference']['path']).read_text()
 assert all(r['target_text'].endswith(model_io.JSON_CALL_STOP_SUFFIXES[0]) for r in selected)
 assert hashlib.sha256(full.encode()).hexdigest()==s['source_content_sha256']
 packet={'source_id':sid,'family':s['family'],'goal':state['goal']['request'],'source_content':full,
 'source_reference':s['content_reference'],'run_root':str(root),
 'rows':[{'sample_id':r['sample_id'],'authority':r['label_authority'],'checkpoint':r['candidate_checkpoint_id'],
  'input_sha256':hashlib.sha256(r['input_text'].encode()).hexdigest(),'target_sha256':hashlib.sha256(r['target_text'].encode()).hexdigest(),
  'answer':model_io.parse_model_command(r['target_text'][:-len(model_io.JSON_CALL_STOP_SUFFIXES[0])]).arguments.get('text'),
  'raw_target':r['target_text']} for r in selected]}
 write_once(D/'packets'/f'{sid}.json',packet)
write_once(D/'SOURCE_MANIFEST.json',m)
print('frozen 128 existing rows, per-source unchanged materials and all targets')

"""One current production Planner request through the native adapter; no role data."""
from dataclasses import fields
from datetime import datetime, timezone
from pathlib import Path
import hashlib
import json
import sys
import time

ROOT = Path('/home/chase/GitHub/RWKV-LH')
sys.path.insert(0, str(ROOT))
from rwkv_lh.goal_loop_protocol import GoalPlanRequest, RollingGoalPlan
from rwkv_lh.supervisor_openai import SupervisorAPISettings, OpenAICompatibleSupervisorClient

ROUND = ROOT / 'data/experiments/VLLM_RWKV_SUPERVISOR_ADAPTER_R1_20260907'
ROUND.mkdir(parents=True, exist_ok=True)
OUTPUT = ROUND / 'PRODUCTION_PLANNER_NATIVE_PROBE.json'
assert not OUTPUT.exists(), OUTPUT
SOURCE = ROOT / 'data/experiments/ZERO_STATE_AGENT_BASELINE_R1_20260907/PLANNER_WIRE_DIAGNOSTIC_R2/SUP-9bb02f7be600485aaedc.reconstructed_body.json'
original = json.loads(SOURCE.read_text())
canonical = json.loads(original['messages'][1]['content'])
values = {field.name: canonical[field.name] for field in fields(GoalPlanRequest) if field.name in canonical}
values['immutable_request'] = canonical['current_requirement']
values['recent_action_facts'] = tuple(values.get('recent_action_facts') or ())
request = GoalPlanRequest(**values)
assert request.to_dict() == canonical
MODEL = 'rwkv7-g1j-13.3b-zero-state-capability-ctx16384'
record = {
    'at': datetime.now(timezone.utc).isoformat(),
    'purpose': 'Authorized native transport/budget validation using a preserved production request; not Agent score or training data',
    'source_sha256': hashlib.sha256(SOURCE.read_bytes()).hexdigest(),
    'script_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    'canonical_request': request.to_dict(), 'events': [], 'http_requests': [],
    'source_module_sha256': {name: hashlib.sha256((ROOT/name).read_bytes()).hexdigest()
                            for name in ('rwkv_lh/supervisor_openai.py', 'rwkv_lh/runtime/supervisor_vllm_rwkv.py', 'rwkv_lh/goal_loop_protocol.py')},
}


class RecordingClient(OpenAICompatibleSupervisorClient):
    def _post_completion(self, endpoint, body):
        assert endpoint == self.settings.base_url + '/completions'
        token_request = {'model': self.model_name, 'prompt': body['prompt'], 'add_special_tokens': True}
        token_response = self._session().post('http://127.0.0.1:29613/tokenize', json=token_request, timeout=20)
        token_response.raise_for_status()
        token_data = token_response.json()
        count = token_data['count']
        assert count + body['max_tokens'] <= 16384, 'Request exceeds deployed model context; do not truncate inputs'
        item = {'endpoint': endpoint, 'request_body': body, 'tokenization': token_data,
                'context_limit': 16384, 'requested_total_context': count + body['max_tokens']}
        record['http_requests'].append(item)
        result = super()._post_completion(endpoint, body)
        item.update(http_status=result.status_code, response_text=result.text,
                    response_sha256=hashlib.sha256(result.content).hexdigest())
        return result


settings = SupervisorAPISettings(
    base_url='http://127.0.0.1:29613/v1', api_key='local-no-auth',
    model=MODEL, stage_checker_model=MODEL, backend_profile='vllm-rwkv-native',
    max_plan_tokens=8192, read_timeout_seconds=240,
    retry_attempts=1, semantic_repair_attempts=0, plan_cache_enabled=False,
    fallback_models=(),
)
settings.validate()
record['settings'] = settings.public_dict()
client = RecordingClient(settings, audit_hook=record['events'].append)
start = time.monotonic()
try:
    record['health'] = client.health()
    patch = client.plan_goal_patch(request)
    record['planner_contract_passed'] = True
    record['patch'] = patch.to_dict()
    plan = RollingGoalPlan(goal_digest=request.goal_digest)
    assert plan.to_model_dict() == request.active_plan
    plan.apply_goal_patch(patch)
    record['rolling_plan_validation_passed'] = True
    record['in_memory_plan'] = plan.to_model_dict()
    record['step_count'] = len(patch.add_steps)
except Exception as exc:
    record['error_type'] = type(exc).__name__
    record['error'] = str(exc)
finally:
    client.close()
    record['elapsed_seconds'] = round(time.monotonic() - start, 3)
    OUTPUT.write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({key: record.get(key) for key in (
    'planner_contract_passed', 'rolling_plan_validation_passed', 'step_count',
    'error_type', 'error', 'elapsed_seconds')}, ensure_ascii=False))

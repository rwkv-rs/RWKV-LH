from pathlib import Path
import sys,json,hashlib,subprocess,time,os
root=Path('/home/chase/GitHub/RWKV-LH');p=root/'data/experiments/RWKV_DUAL_COLLECTION_R10_20260916';frozen=p/'runtime'
sys.path.insert(0,str(frozen))
from rwkv_lh.runtime.settings import load_local_env
from rwkv_lh.collection_conversion import export_boundaries
load_local_env(root/'.env.local')
def put(value):
 tmp=p/'STATUS.pending';tmp.write_text(json.dumps(value,ensure_ascii=False,indent=2));os.replace(tmp,p/'STATUS.json')
def main():
 reg=json.loads((p/'REGISTRATION.json').read_text())
 for name,digest in reg['runtime_files'].items():
  if hashlib.sha256((frozen/name).read_bytes()).hexdigest()!=digest:raise ValueError('frozen runtime changed: '+name)
 for name,digest in reg['inputs'].items():
  if hashlib.sha256((p/name).read_bytes()).hexdigest()!=digest:raise ValueError('frozen input changed: '+name)
 cmd=[sys.executable,str(frozen/'scripts/run_collection_queue.py'),'--queue',str(p/'queue.sqlite3')]
 if not (p/'queue.sqlite3').exists():subprocess.run(cmd+['--inventory',str(p/'inventory.jsonl')],check=True)
 put({'phase':'collecting','prepared_tasks':5,'target_tasks':30000,'teacher_calls':0,'training_rows':0,'started':time.time()})
 result=subprocess.run(cmd+['--run','--replicas',str(p/'replicas.json'),'--concurrency','4','--max-hours','36'])
 results=[]
 for line in (p/'inventory.jsonl').read_text().splitlines():
  item=json.loads(line);run=Path(item['job']['output_dir']);receipt=run/'COLLECTION_RECEIPT.json'
  if not receipt.exists():continue
  r=json.loads(receipt.read_text())['result']; row={'source_id':item['source_id'],'result':r}
  if r.get('trace_complete'):
   export=run.parent/'boundaries'
   if not export.exists():export_boundaries(run/'execution',reg['model_sha256'],export)
   first=json.loads((export/'boundaries.jsonl').open().readline())
   if first['input_token_ids']!=json.loads((run.parent/'PREVIEW.json').read_text())['input_token_ids']:raise ValueError('initial input mismatch')
   row['boundaries_verified']=True
  results.append(row)
 (p/'RESULTS.json').write_text(json.dumps(results,ensure_ascii=False,indent=2))
 put({'phase':'batch_recorded_waiting_for_more_bound_tasks' if result.returncode==0 else 'stopped_review_required','exit':result.returncode,'recorded_tasks':len(results),'prepared_tasks':5,'target_tasks':30000,'bulk_campaign_complete':False,'teacher_calls':0,'training_rows':0,'acceptance':'pending_independent_review','ended':time.time()})
 return result.returncode
if __name__=='__main__':raise SystemExit(main())

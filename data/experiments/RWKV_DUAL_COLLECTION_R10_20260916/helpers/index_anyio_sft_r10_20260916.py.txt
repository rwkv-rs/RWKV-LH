from pathlib import Path
import json, hashlib
root=Path('/home/chase/GitHub/RWKV-LH');out=root/'data/experiments/RWKV_DUAL_COLLECTION_R10_20260916'
seen=set();rows=[]
for p in sorted((root/'data/experiments/RWKV_SFT_BACKGROUND_R8_20260915').glob('Code_Agent_part-*/source.jsonl')):
 for raw in p.open('rb'):
  r=json.loads(raw)
  users=[m.get('content','') for m in r.get('messages',[]) if m.get('role')=='user']
  if len(users)!=1 or not isinstance(users[0],str):continue
  u=users[0]
  if '<pr_description>' not in u or '</pr_description>' not in u:continue
  desc=u.split('<pr_description>')[1].split('</pr_description>')[0]
  if 'anyio' not in desc.lower():continue
  digest=hashlib.sha256(' '.join(desc.split()).encode()).hexdigest()
  if digest in seen:continue
  seen.add(digest); target=out/'candidates'/r['uuid'];target.mkdir(parents=True,exist_ok=True)
  (target/'source.json').write_bytes(raw)
  (target/'description.txt').write_text(desc)
  rows.append({'id':r['uuid'],'title':desc.strip().splitlines()[0],'source':str(target/'source.json'),'description_sha256':digest})
(out/'ANYIO_CANDIDATES.json').write_text(json.dumps(rows,indent=2,ensure_ascii=False))
print('Distinct whitespace-normalized AnyIO descriptions',len(rows))
for r in rows:print(r['id'],r['title'])

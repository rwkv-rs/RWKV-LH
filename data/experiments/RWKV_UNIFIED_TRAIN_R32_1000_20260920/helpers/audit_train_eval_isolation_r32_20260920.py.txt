from pathlib import Path
import json,hashlib,sys
R=Path('/home/chase/GitHub/RWKV-LH');sys.path.insert(0,str(R))
from rwkv_lh.role_trace_artifacts import byte_5gram_cosine
D=R/'data/experiments/RWKV_UNIFIED_TRAIN_R32_1000_20260920'
P=R/'data/experiments/RWKV_UNIFIED_TRAIN_R32_1000_20260920'
rows=json.loads((P/'REVIEWED_ROWS.json').read_text())['rows']
sources=json.loads((P/'FREEZE_REGISTRATION.json').read_text())['sources']
policy=json.loads((D/'EVALUATION_REGISTRATION.json').read_text())
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
records=[]
for case in policy['cases']:
 for name,h in case['files'].items():
  p=Path(case['workspace'])/name;assert sha(p)==h;text=p.read_bytes().decode();encoded=json.dumps(text,ensure_ascii=False)[1:-1]
  overlap=[r['sample_id'] for r in rows if r['source_content_sha256']==h or text in r['input_text'] or encoded in r['input_text'] or text in r['target_text'] or encoded in r['target_text']]
  assert not overlap,(case['id'],name,overlap)
  nearest=max(({'id':s['source_id'],'cosine':byte_5gram_cosine(text,Path(s['content_reference']['path']).read_bytes().decode())} for s in sources),key=lambda x:x['cosine'])
  assert nearest['cosine']<0.9,(case['id'],name,nearest)
  records.append({'case':case['id'],'file':name,'sha256':h,'whole_file_overlap':False,'nearest_training_source':nearest})
out={'training_rows':len(rows),'training_rows_sha256':sha(P/'REVIEWED_ROWS.json'),'registered_evaluation_sha256':sha(D/'EVALUATION_REGISTRATION.json'),
     'algorithm':'UTF8 byte5gram cosine','threshold':0.9,'passed':True,'files':records,'final_holdout_read':False,
     'scope':'Same12 previously observed development tasks, excluded from these1000 StateTune rows. Not a claim about base pretraining exposure or fresh final acceptance.'}

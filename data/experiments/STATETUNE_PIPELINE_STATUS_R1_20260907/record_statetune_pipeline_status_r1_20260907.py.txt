"""Record the documentation-only StateTune inventory and completed validation."""

from __future__ import annotations

import ast
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess


ROOT = Path("/home/chase/GitHub/RWKV-LH")
ROUND_ID = "STATETUNE_PIPELINE_STATUS_R1_20260907"
OUTPUT = ROOT / "data" / "experiments" / ROUND_ID
DOCUMENT = ROOT / "docs" / "STATETUNE_DATA_PIPELINE_STATUS.zh-CN.md"


def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git(*arguments: str) -> str:
    return subprocess.check_output(
        ["git", *arguments], cwd=ROOT, text=True
    ).strip()


def main() -> None:
    assert os.environ.get("WSL_DISTRO_NAME") == "UbuntuRecovered"
    planned = ROOT / "rwkv_lh/goal_state_protocols/role_trace_dataset_v1.py"
    assert not planned.exists(), "Pipeline status needs reinspection: extractor now exists"
    assert not git("diff", "HEAD", "--name-only", "--", "rwkv_lh/", "scripts/", "tests/")
    test_source = ROOT / "tests/test_retired_role_data_chain.py"
    tree = ast.parse(test_source.read_text(encoding="utf-8"))
    retired_stems = next(
        ast.literal_eval(node.value)
        for node in tree.body
        if isinstance(node, ast.Assign)
        and any(isinstance(target, ast.Name) and target.id == "RETIRED_SCRIPT_STEMS"
                for target in node.targets)
    )
    retired = [
        {
            "path": f"scripts/{stem}.py",
            "source_present": (ROOT / "scripts" / f"{stem}.py").exists(),
            "bytecode_present": [
                str(path.relative_to(ROOT))
                for path in (ROOT / "scripts").rglob(f"{stem}*.pyc")
            ],
        }
        for stem in retired_stems
    ]
    assert all(not item["source_present"] and not item["bytecode_present"] for item in retired)
    links = []
    for target in re.findall(r"\]\(([^)]+)\)", DOCUMENT.read_text(encoding="utf-8")):
        if "://" in target or target.startswith("#"):
            continue
        path = (DOCUMENT.parent / target.split("#", 1)[0]).resolve()
        assert path.is_relative_to(ROOT), target
        assert path.exists(), target
        links.append(str(path.relative_to(ROOT)))
    test_log = OUTPUT / "full_pytest.txt"
    test_text = test_log.read_text(encoding="utf-8")
    match = re.search(r"(\d+) passed in ([0-9.]+)s", test_text)
    assert match, "Full test run has not passed"
    assert not re.search(r"\d+ (?:failed|skipped|errors?|xfailed|xpassed)\b", test_text)
    passed, duration = int(match.group(1)), float(match.group(2))
    history = json.loads((OUTPUT / "HISTORICAL_GITHUB_SOURCES.json").read_text(encoding="utf-8"))
    assert history["verified_source_count"] == 15
    assert all(item["remote_blob_matches_local"] for item in history["source_files"])
    inputs = [
        "AGENTS.md", "README.md", "docs/HANDOFF.zh-CN.md",
        "docs/G1J_UNIFIED_PROTOCOL_ITERATION_PLAN.zh-CN.md",
        str(DOCUMENT.relative_to(ROOT)),
        "rwkv_lh/goal_state_protocols/selector_intent_v4.py",
        "rwkv_lh/goal_state_protocols/executor_args_v4.py",
        "rwkv_lh/goal_state_protocols/auditor_step_v3.py",
        "rwkv_lh/goal_state_protocols/finalizer_answer.py",
        "rwkv_lh/goal_state_protocols/auditor_final.py",
        "rwkv_lh/store.py", "rwkv_lh/model_session.py",
        "scripts/run_rwkv_e2e_benchmark.py", "tests/test_retired_role_data_chain.py",
        "data/experiments/PROTOCOL_DATA_CHAIN_UNIFICATION_R1_20260907/ROUND_ANALYSIS.zh-CN.md",
        "data/experiments/PROTOCOL_DATA_CHAIN_UNIFICATION_R1_20260907/LEGACY_ROLE_CHAIN_REMOVAL_RESULT.json",
        "data/experiments/ZERO_STATE_AGENT_BASELINE_R1_20260907/TRACE_READINESS.zh-CN.md",
        "data/experiments/ZERO_STATE_AGENT_BASELINE_R1_20260907/TRAINING_ROUND_ACCOUNTING_REVIEW.zh-CN.md",
        "data/experiments/ZERO_STATE_AGENT_BASELINE_R1_20260907/R2_A_RUN_REPORT.zh-CN.md",
    ]
    branch = git("branch", "--show-current")
    evidence = {
        "round_id": ROUND_ID,
        "date": "2026-09-07",
        "environment": "WSL UbuntuRecovered",
        "scope": "Document current StateTune pipeline status and recorded project lessons; publish current branch at user request.",
        "agent_metrics": {"strict": None, "completed": None, "mutation": None, "termination_reasons": None, "new_evaluation_run": False},
        "role_training_runs": 0,
        "new_dataset_versions": 0,
        "holdout_read_or_run": False,
        "confirmation_rescored": False,
        "runtime_source_modified": False,
        "git": {
            "local_head_before_documentation_commit": git("rev-parse", "HEAD"),
            "branch": branch,
            "upstream": git("rev-parse", "--abbrev-ref", "@{upstream}"),
            "fetched_upstream_sha": git("rev-parse", "@{upstream}"),
            "preexisting_unpushed_commits": git("rev-list", "--reverse", "@{upstream}..HEAD").splitlines(),
            "publication_authorization": "User: 是的话更新到github上",
        },
        "planned_extractor": {"path": str(planned.relative_to(ROOT)), "exists": False},
        "retired_entrypoints": retired,
        "historical_source_verification": {
            "evidence": "HISTORICAL_GITHUB_SOURCES.json",
            "sha256": sha256(OUTPUT / "HISTORICAL_GITHUB_SOURCES.json"),
            "github_verified_sources": history["verified_source_count"],
            "historical_commit": history["historical_commit"],
            "newer_retired_entrypoints_without_reachable_history": sum(not row["commits_in_all_local_refs"] for row in history["newer_retired_entrypoint_history_lookup"]),
        },
        "documentation_local_links": {"count": len(links), "all_targets_exist": True, "targets": links},
        "validation": {
            "environment_sync": {"command": "uv sync --frozen --extra selector-runtime --extra benchmark-web --group dev", "exit_code": 0, "log": "environment_sync.txt"},
            "browser_install": {"command": ".venv/bin/python -m playwright install chromium", "exit_code": 0, "log": "browser_install.txt"},
            "full_regression": {"command": ".venv/bin/python -m pytest -q tests/", "exit_code": 0, "passed": passed, "skipped": 0, "duration_seconds": duration, "log": "full_pytest.txt"},
        },
        "source_sha256": {path: sha256(ROOT / path) for path in inputs},
        "recorder": {"path": str(Path(__file__).relative_to(ROOT)), "sha256": sha256(Path(__file__))},
    }
    (OUTPUT / "AUDIT.json").write_text(json.dumps(evidence, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    report = f"""# StateTune 数据管线现状与经验整理 R1

轮次：`{ROUND_ID}`。日期：2026-09-07。执行环境：WSL `UbuntuRecovered`。

## Agent 与角色结果

本轮没有运行 Agent 评测，Strict / completed / mutation / 终止原因均无新增指标。没有角色训练、角色评测、新建角色 datasets、confirmation 重评分或 Holdout 读取/运行。完整代码回归不代表模型能力提升。

## 核查结论与改动

当前工作树没有可运行的完整 StateTune 角色数据生成管线。已核对生产代码、scripts、temp 文件清单与现行规范：旧合成生成链已删除；规范指定的 `rwkv_lh/goal_state_protocols/role_trace_dataset_v1.py` 不存在。退休入口防回归测试列出的 {len(retired)} 个旧源码入口及其对应字节码均未残留。五角色共享 builder 与 LongHorizonStore 是现有基础组件，尚不构成完成逐行 State 上下文重建和数据验收的抽取器。

新增 [StateTune 数据管线源码、现状与经验](../../../docs/STATETUNE_DATA_PIPELINE_STATUS.zh-CN.md)，汇总组件、缺口、未来流程和六条有来源的经验；README 与 HANDOFF 增加入口。新文档的 {len(links)} 个本地链接均存在。经验区分既有缺陷证据和待实现的数据/比较要求，未宣称训练增益。

用户进一步明确数据生成管线源码也需上传，因此追加全部可达 Git refs 的历史检查与 GitHub 实际核验。旧五角色 G1J v1 的入口、dataset_contract 核心、verifier、来源登记、旧协议与 tokenizer 共 15 个源文件已在 GitHub 保存，远端大小与 Git blob OID 全部匹配本地历史提交 `3f23a6a6fe3e048e3142f43aae4938f92ea5db2c`；新文档给出完整源码入口和依赖关系，记录见 [HISTORICAL_GITHUB_SOURCES.json](HISTORICAL_GITHUB_SOURCES.json)。当前组件随现有分支发布，旧实现按既有清理规范从历史访问，未恢复旧可执行入口。

10 个较新旧入口按原路径在全部已抓取本地可达 refs 中没有历史命中，只有删除清单记录；不能宣称这些源码已上传，也不能由 SHA 重建内容。其他位置是否有副本未确认。历史源码可访问不证明旧数据仍可取得、旧链当前可运行或符合当前生产协议。

核查没有改动生产源码、测试或运行参数。协议 builder、State 生命周期和 checkpoint 导出相关代码、清理报告、trace 可用性报告、轮次对账与现行文档的 SHA-256 均登记于 [AUDIT.json](AUDIT.json)。生成记录的临时脚本按规范位于 temp，逐字节快照保存为 `record_statetune_pipeline_status_r1_20260907.py.txt`，生产代码不依赖它。

## 验证

环境同步与 Chromium 安装均退出 0。完整命令 `.venv/bin/python -m pytest -q tests/`：**{passed} passed in {duration:.2f}s，0 skipped**，包含现有 Torch / State 注入与浏览器回归；原始日志见 [full_pytest.txt](full_pytest.txt)。本轮只整理文档，无代码修复，因此不新增镜像实现的测试或虚构先失败记录。

用户在本会话明确要求更新到 GitHub。本地当前分支 `{branch}` 已跟踪 origin 同名分支，抓取后无远端独有提交，已有 {len(evidence['git']['preexisting_unpushed_commits'])} 个本地提交尚未推送。本轮文档提交追加于其后，并从本地推送该分支；不修改 main。AUDIT 记录推送前提交身份，发布成功与远端 HEAD 核验在会话中报告，本文不提前声称推送成功。

## 证据指纹

本轮文件与引用源的完整 SHA-256 见 [AUDIT.json](AUDIT.json)；本轮工件统一校验入口为 [SHA256SUMS](SHA256SUMS)。

    (OUTPUT / "REPORT.zh-CN.md").write_text(report, encoding="utf-8")
    (OUTPUT / "record_statetune_pipeline_status_r1_20260907.py.txt").write_bytes(Path(__file__).read_bytes())

    artifacts.extend(ROOT / path for path in inputs)
    (OUTPUT / "SHA256SUMS").write_text(
        "".join(f"{sha256(path)}  {path.relative_to(ROOT)}\n" for path in sorted(set(artifacts))),
        encoding="utf-8",
    )
    print(json.dumps({"round_id": ROUND_ID, "passed": passed, "skipped": 0, "local_links": len(links), "document_sha256": sha256(DOCUMENT), "report_sha256": sha256(OUTPUT / "REPORT.zh-CN.md")}, ensure_ascii=False))

    main()

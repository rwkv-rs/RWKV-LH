"""Resume one recorded RWKV task with one budgeted DeepSeek advice request.

Run with PYTHONPATH pointing at the parent's frozen production source. This
orchestrator does not replace tool selection, parameters, or final output.
"""
import argparse
from dataclasses import asdict
import json
import os
from pathlib import Path
import subprocess
import sys

from rwkv_lh.assisted_agent import AssistedJob, load_parent, continue_assisted
from rwkv_lh.offline_teacher_api import DeepSeekTeacher, write_json
from rwkv_lh.runtime.settings import RuntimeSettings, direct_agent_settings
from rwkv_lh.summary_advice import request_advice


class BudgetedAdviceClient:
    def __init__(self, teacher):
        self.teacher = teacher

    def _request_json(self, *, system_prompt, request_payload, **kwargs):
        value, self.identity = self.teacher.complete(
            system_prompt, json.dumps(request_payload, ensure_ascii=False))
        return value


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--registration', type=Path)
    parser.add_argument('--worker', action='store_true')
    args = parser.parse_args()
    if args.worker:
        payload = json.load(sys.stdin)
        continue_assisted(AssistedJob(**payload['job']),
            settings=RuntimeSettings(**payload['settings']),
            advice=payload['advice'], advice_model=payload['advice_model'])
        return
    config = json.loads(args.registration.read_text())
    job = AssistedJob(**config['job'])
    if job.mode != 'advice':
        raise ValueError('only advice; no takeover')
    previous, result, state, workspace = load_parent(job)
    root = Path(config['audit_directory'])
    root.mkdir(parents=True, exist_ok=False)
    if Path(job.output_dir).exists():
        raise ValueError('continuation output already exists')
    client = BudgetedAdviceClient(DeepSeekTeacher(**config['api']))
    advice = request_advice(client, state.goal, {}, result['final'] or '', job.task_id,
        job.advice_max_tokens, tool_observations=[a.to_dict() for a in state.actions.values()])
    write_json(root / 'ADVICE.json', {'advice': advice, 'identity': client.identity})
    settings = direct_agent_settings(RuntimeSettings(**config['settings']))
    payload = {'job': asdict(job), 'settings': asdict(settings), 'advice': advice,
               'advice_model': client.identity['model']}
    command = ['unshare', '--user', '--map-root-user', '--mount', '--propagation', 'private',
               sys.executable, str(Path(__file__).resolve()), '--worker']
    from rwkv_lh.assisted_agent import _run_worker
    completed = _run_worker(command, json.dumps(payload), job.max_seconds)
    (root / 'WORKER.stdout').write_text(completed.stdout)
    (root / 'WORKER.stderr').write_text(completed.stderr)
    write_json(root / 'RESULT.json', {'exit_code': completed.returncode,
        'strong_requests': 1, 'takeover': False, 'training_started': False})
    if completed.returncode:
        raise RuntimeError('advice continuation failed; see audited worker stderr')
    output = Path(job.output_dir)
    write_json(output / 'execution/SOURCE_PURPOSE.json',
               {'source_purpose': config['source_purpose']})
    if config.get('acceptance_command'):
        with (root / 'ACCEPTANCE.log').open('w') as log:
            verdict = subprocess.run(config['acceptance_command'], stdout=log, stderr=log)
        write_json(root / 'ACCEPTANCE_RESULT.json', {'exit_code': verdict.returncode})


if __name__ == '__main__':
    main()

from pathlib import Path
import collections
import hashlib
import json

R = Path('/home/chase/GitHub/RWKV-LH')
D = R / 'data/experiments/RWKV_UNIFIED_DATA_EXPANSION_R23_20260919'
packages = [R / 'data/datasets/rwkv_direct_unified_corrections_v6',
            R / 'data/experiments/RWKV_REAL_CODING_DATA_R20_20260917/frozen_increment',
            R / 'data/experiments/RWKV_CODEX_FREEZE_R22_20260919/frozen_increment']
packages += sorted(p.parent / 'frozen_increment' for p in (D / 'batches').glob('*/RESULT.json'))
counts = collections.Counter()
rows = []
for package in packages:
    for line in (package / 'train.jsonl').read_text().splitlines():
        row = json.loads(line)
        target = row['target_text'].lstrip()
        if target.startswith('```json'):
            target = target[len('```json'):].lstrip()
        try:
            value, _ = json.JSONDecoder().raw_decode(target)
            keys = ','.join(sorted(value)) if isinstance(value, dict) else 'not_object'
        except (ValueError, TypeError):
            keys = 'not_standalone_json'
        instruction = ('"function" for its name and "params" for its complete parameters'
                       in row['input_text'])
        counts[(keys, instruction)] += 1
        rows.append({'sample_id': row['sample_id'], 'package': str(package),
                     'target_sha256': hashlib.sha256(row['target_text'].encode()).hexdigest(),
                     'target_keys': keys, 'input_instructs_function_params': instruction})
out = {'rows': len(rows), 'groups': [{'target_keys': k[0],
        'input_instructs_function_params': k[1], 'count': v} for k, v in sorted(counts.items())],
       'interpretation': 'Lexical input/target consistency audit only. Parser acceptance and executable correction validation do not establish canonical training-label consistency. No labels rewritten and no causal attribution to prior model results.',
       'samples': rows}
(D / 'TARGET_ENVELOPE_AUDIT.json').write_text(json.dumps(out, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({k: v for k, v in out.items() if k != 'samples'}, ensure_ascii=False, indent=2))

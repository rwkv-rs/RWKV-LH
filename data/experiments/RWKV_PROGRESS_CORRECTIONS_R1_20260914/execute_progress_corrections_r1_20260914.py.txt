from pathlib import Path
import sys,json,hashlib
R=Path('/home/chase/GitHub/RWKV-LH');sys.path.insert(0,str(R))
from rwkv_lh import model_io
from rwkv_lh.harness import ActionHarness
from rwkv_lh.schema import GoalState,TaskAction
from rwkv_lh.workspace_snapshot import copy_verified_workspace,tree_identity
D=R/'data/experiments/RWKV_PROGRESS_CORRECTIONS_R1_20260914'
reg=json.loads((D/'REGISTRATION.json').read_text());results=[]
for source in reg['sources']:
 root=Path(source['run_root']);state=json.loads((root/'state_snapshot.json').read_text())
 request_ids={e['request_id'] for e in map(json.loads,(root/'model_trace.jsonl').read_text().splitlines()) if e['type']=='model_session_generation_returned'}
 actions=sorted((a for a in state['actions'].values() if a['request_id'] in request_ids),key=lambda a:a['sequence'])
 calls=sorted((root/'tool_snapshots').glob('*/call.json'));assert len(actions)==len(calls)
 for action,path in zip(actions,calls):
  call=json.loads(path.read_text());assert action['action_type']==call['action_type'] and action['arguments']==call['arguments']
 matched=[(a,p) for a,p in zip(actions,calls) if a['request_id']==source['actual']['request_id']];assert len(matched)==1
 action,call=matched[0];before=call.parent/'before'
 folder=D/'runs'/source['id'];workspace=folder/'workspace';copy_verified_workspace(before,workspace)
 candidate=json.loads((folder/'CANDIDATE.json').read_text());command=model_io.parse_model_command(candidate['raw_output'])
 goal=GoalState.create(request=state['goal']['request'],constraints=(),workspace_root=str(workspace))
 result=ActionHarness().execute(TaskAction(command.name,command.arguments),goal).to_dict()
 record={'id':source['id'],'source_checkpoint':source['checkpoint'],'source_action_id':action['action_id'],'snapshot':str(before.relative_to(root)),'snapshot_tree':tree_identity(before),'raw_target':candidate['raw_output'],'tool_result':result,'training_admitted':False}
 (folder/'EXECUTION.json').write_text(json.dumps(record,ensure_ascii=False,indent=2)+'\n');results.append(record)
 print(source['id'],command.name,'success',result['success'],'exit',result.get('exit_code'),'error',result.get('error'),flush=True)
(D/'EXECUTIONS.json').write_text(json.dumps(results,ensure_ascii=False,indent=2)+'\n')

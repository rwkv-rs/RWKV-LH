from pathlib import Path
from dataclasses import replace
import sys,json,hashlib
R=Path('/home/chase/GitHub/RWKV-LH');sys.path.insert(0,str(R))
from rwkv_lh.correction_review import build_review_packet,validate_review,REVIEW_INSTRUCTION,REVIEW_SCHEMA
from rwkv_lh.strong_session import AuditedStrongClient
from rwkv_lh.supervisor_openai import SupervisorAPISettings
from rwkv_lh.runtime.settings import load_local_env
from rwkv_lh.command_corrections import validate_command_correction
from rwkv_lh import model_io
D=R/'data/experiments/RWKV_PROGRESS_CORRECTIONS_R1_20260914'
reg=json.loads((D/'REGISTRATION.json').read_text());load_local_env(R/'.env.local')
settings=replace(SupervisorAPISettings.from_env(),retry_attempts=1,semantic_repair_attempts=0,fallback_models=(),plan_cache_enabled=False,planner_request_options={'thinking':{'type':'disabled'}},read_timeout_seconds=120)
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def save(p,x):p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(x,ensure_ascii=False,indent=2)+'\n')
save(D/'REVIEW_REGISTRATION.json',{'script_sha256':sha(Path(__file__)),'calls':4,'max_output_tokens':4096,'settings':settings.public_dict(),'system':REVIEW_INSTRUCTION,'schema':REVIEW_SCHEMA,'external_command_gate':{'exit_code':1,'output':['ValueError: value must be positive','Ran 1 test']},'quality':'Next test action is not a completed repair and expected tool failure is not a protocol error. Do not inject future test result into review or training input.'})
for source in reg['sources']:
 folder=D/'runs'/source['id'];candidate=json.loads((folder/'CANDIDATE.json').read_text());packet=build_review_packet(actual_rwkv_input=source['actual']['input_text'],candidate=candidate['raw_output'])
 target=candidate['raw_output']+model_io.JSON_CALL_STOP_SUFFIXES[0];reviews=[]
 for repeat in (1,2):
  dest=folder/f'review-{repeat}';dest.mkdir()
  def audit(event):
   with (dest/'strong_trace.jsonl').open('a') as stream:stream.write(json.dumps(dict(event),ensure_ascii=False)+'\n')
  client=AuditedStrongClient(settings,audit_hook=audit);save(dest/'REQUEST.json',packet)
  try:
   value=client._request_json(phase='state_tune_target_review',run_id=source['id'],request_digest=hashlib.sha256(json.dumps(packet).encode()).hexdigest(),system_prompt=REVIEW_INSTRUCTION,request_payload=packet,schema=REVIEW_SCHEMA,max_tokens=4096)
   save(dest/'RAW_REVIEW.json',value);v=validate_review(packet,value);save(dest/'VALIDATED_REVIEW.json',v)
   reviews.append({'reviewer':settings.model+f'/independent-context-{repeat}','accepted':v['accepted'],'visible_evidence_only':True,'target_sha256':hashlib.sha256(target.encode()).hexdigest(),'input_sha256':packet['input_sha256'],'review_path':str(dest/'RAW_REVIEW.json'),'review_sha256':sha(dest/'RAW_REVIEW.json')})
  except Exception as exc:save(dest/'ERROR.json',{'error':type(exc).__name__+': '+str(exc)})
  finally:client.close()
 save(folder/'REVIEWS.json',reviews)
 if len(reviews)==2 and all(r['accepted'] for r in reviews):
  validation=dict(run_root=source['run_root'],checkpoint_id=source['checkpoint'],target_text=target,model_sha256='559371f5b9aef13189ae54b345ac096af4ad2b689996c05d89de687612b3ae65',source_files=source['source_files'],reviews=reviews,expected_exit_code=1,expected_output=['ValueError: value must be positive','Ran 1 test'],output=str(folder/'validation'))
  save(folder/'VALIDATION_REGISTRATION.json',validation);value=validate_command_correction(**validation)
  print(source['id'],value['status'],flush=True)
 else:print(source['id'],'review rejected',flush=True)

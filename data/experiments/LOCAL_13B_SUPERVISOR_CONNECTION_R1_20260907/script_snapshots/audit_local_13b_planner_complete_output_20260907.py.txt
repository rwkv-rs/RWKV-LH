def attempt(fn):
    try:
        value = fn()
        return {'passed': True}, value
    except Exception as exc:
        return {'passed': False, 'error_type': type(exc).__name__, 'error': str(exc)}, None

def tokenize(text, model):
    body = {'model': model, 'prompt': text, 'add_special_tokens': False, 'return_token_strs': False}
    raw = json.dumps(body, ensure_ascii=False).encode()
    request = urllib.request.Request('http://127.0.0.1:29613/tokenize', data=raw, headers={'Content-Type': 'application/json'}, method='POST')
    result = {'request_sha256': hashlib.sha256(raw).hexdigest(), 'text_sha256': hashlib.sha256(text.encode()).hexdigest(), 'chars': len(text), 'add_special_tokens': False}
    try:
        with urllib.request.urlopen(request, timeout=20) as response:
            payload = response.read(); data = json.loads(payload)
            result.update(http_status=response.status, response_sha256=hashlib.sha256(payload).hexdigest(), response=data)
    except urllib.error.HTTPError as exc:
        payload = exc.read(); result.update(http_status=exc.code, error=payload.decode(errors='replace')[:2000])
    except (OSError, ValueError) as exc:
        result.update(http_status=None, error_type=type(exc).__name__, error=str(exc))
    return result

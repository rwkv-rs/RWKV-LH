"""Create deployment identity manifests locally from Git-free file inspection."""
from pathlib import Path
import hashlib
import json
import sys

ROOT=Path('/home/chase/GitHub/RWKV-LH')
sys.path.insert(0,str(ROOT))
from rwkv_lh.state_router.local_backend import ENGINE_SOURCE_MANIFEST_SCHEMA

ROUND=ROOT/'data/experiments/ZERO_STATE_AGENT_BASELINE_R1_20260907'
directory=ROOT/'data/runtime/manifests'
directory.mkdir(parents=True,exist_ok=True)
records=json.loads((ROUND/'UPLOAD_ONLY_ENGINE_FILES.json').read_text())
frozen=json.loads((ROUND/'FROZEN_EXECUTION_MANIFEST.json').read_text())
result=[]
for name,files in records.items():
    # The revision label comes from the already recorded local run manifest.
    # The current server identity is the complete uploaded file set + SHA.
    revision=frozen['remote_deployment']['engines'][name]['revision']
    value={'schema_version':ENGINE_SOURCE_MANIFEST_SCHEMA,
           'engine_root':str(ROOT/'data/runtime/engines'/name),
           'engine_revision':revision,'source_root':'.','files':files}
    path=directory/(name+'.source.json')
    path.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')
    result.append({'name':name,'path':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
                   'files':len(files),'bytes':sum(item['bytes'] for item in files)})
(ROUND/'UPLOAD_ONLY_MANIFESTS.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result,indent=2))

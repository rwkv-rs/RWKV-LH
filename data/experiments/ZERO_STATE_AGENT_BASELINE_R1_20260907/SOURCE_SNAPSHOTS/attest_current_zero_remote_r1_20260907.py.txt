"""Read exact current-zero deployment identities; no generation or environment dump."""
from pathlib import Path
import hashlib
import json
import subprocess

ROOT = Path('/home/chase/GitHub/RWKV-LH')
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
def run(*args):
    return subprocess.check_output(args, text=True).strip()

units={}
for role in ('executor','selector'):
    unit=f'rwkv-lh-current-zero-{role}-20260907.service'
    values=run('systemctl','--user','show',unit,'--property=ActiveState,SubState,MainPID').splitlines()
    units[role]=dict(item.split('=',1) for item in values)
    assert units[role]['ActiveState']=='active' and int(units[role]['MainPID'])>0
artifacts={str(p.relative_to(ROOT)):sha(p) for p in (
    ROOT/'data/runtime/launchers/executor_current_zero.sh',
    ROOT/'data/runtime/launchers/selector_current_zero.sh',
    ROOT/'data/runtime/selector/current_decoder_manifest.json')}
models={}
for size in ('13.3b','2.9b'):
    folder=ROOT/f'data/models/rwkv7-g1j-{size}-vllm-v1'
    manifest=json.loads((folder/'manifest.json').read_text())
    stat=(folder/'model.safetensors').stat()
    models[size]={'manifest_sha256':sha(folder/'manifest.json'),
                  'config_sha256':sha(folder/'config.json'),
                  'safetensors':{'bytes':stat.st_size,'inode':stat.st_ino,'mtime_ns':stat.st_mtime_ns},
                  'declared_source':manifest['source'], 'declared_output':manifest['output']}
engines={}
for name in ('vllm-rwkv-67f0c5996c50','vllm-rwkv-67f0c5996c50-selector-clean'):
    folder=ROOT/'data/runtime/engines'/name
    files=run('git','-C',str(folder),'ls-files','--cached','--others','--exclude-standard','--','vllm').splitlines()
    sources={p:sha(folder/p) for p in files if (folder/p).is_file() and Path(p).suffix in {'.py','.cu','.cuh','.h','.cpp','.json'}}
    engines[name]={'revision':run('git','-C',str(folder),'rev-parse','HEAD'),
                   'source_sha256':sources,
                   'diff_sha256':hashlib.sha256(subprocess.check_output(['git','-C',str(folder),'diff','--binary','HEAD','--','vllm'])).hexdigest()}
project_sources={str(path.relative_to(ROOT)):sha(path) for path in sorted((ROOT/'rwkv_lh').rglob('*.py'))}
print(json.dumps({'units':units,'artifacts':artifacts,'models':models,'engines':engines,
                  'project_sources':project_sources},sort_keys=True,indent=2))

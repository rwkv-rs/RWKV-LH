def build():
    for task_id, (request, entrypoint, contract, capabilities) in CONTRACTS.items():
        task = {'task_id': task_id, 'level': 'project', 'user_request': request,
                'capabilities': capabilities, 'workspace_files': [{'path':'README.md','content':contract}], 'workspace_generators':[]}
        (ROOT/task_id/'task.json').write_text(json.dumps(task, ensure_ascii=False, indent=2)+'\n')


def validate():
    results = []
    runtime = ROOT/'author_validation'
    runtime.mkdir(exist_ok=True)
    temporary = runtime/'temporary'
    temporary.mkdir(exist_ok=True)
    environment = dict(os.environ, TMPDIR=str(temporary), TEMP=str(temporary), TMP=str(temporary), PYTHONDONTWRITEBYTECODE='1')
    for task_id, (_, entrypoint, _, _) in CONTRACTS.items():
        task_root = ROOT/task_id
        for variant in ('reference','mutant'):
            workspace = runtime/task_id/variant
            workspace.mkdir(parents=True, exist_ok=True)
            shutil.copy2(task_root/'reference'/entrypoint, workspace/entrypoint)
            if variant == 'mutant':
                mutation = json.loads((task_root/'mutation.json').read_text())
                path = workspace/mutation['file']
                text = path.read_text()
                assert text.count(mutation['replace']) == 1
                path.write_text(text.replace(mutation['replace'],mutation['with']))
            result = subprocess.run([sys.executable, str(task_root/'private_verify.py'), str(workspace)],
                                    cwd=workspace, env=environment,capture_output=True,text=True,timeout=120)
            (runtime/f'{task_id}_{variant}.log').write_text(result.stdout+result.stderr)
            expected = result.returncode == 0 if variant == 'reference' else result.returncode != 0
            results.append({'task_id':task_id,'variant':variant,'returncode':result.returncode,'expectation_met':expected,
                            'candidate_sha256':hashlib.sha256((workspace/entrypoint).read_bytes()).hexdigest(),
                            'verifier_sha256':hashlib.sha256((task_root/'private_verify.py').read_bytes()).hexdigest(),
                            'log':str((runtime/f'{task_id}_{variant}.log').relative_to(ROOT))})
            print(task_id,variant,result.returncode,'OK' if expected else 'FAILED')
            shutil.rmtree(workspace)
        (runtime/task_id).rmdir()
    (runtime/'RESULTS.json').write_text(json.dumps(results, indent=2)+'\n')
    return all(row['expectation_met'] for row in results)


def manifest():
    paths = [path for path in ROOT.rglob('*') if path.is_file() and path.name not in {'MANIFEST.json','SHA256SUMS'} and '__pycache__' not in path.parts]
    entries = [{'path':str(path.relative_to(ROOT)), 'sha256':hashlib.sha256(path.read_bytes()).hexdigest(), 'bytes':path.stat().st_size} for path in sorted(paths)]
    value = {'schema_version':'rwkv-lh.real-project-cli-data-authoring.v1',
             'source':'owner-authorized authored development benchmark',
             'author_script':{'path':str(Path(__file__).resolve()),'sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()},
             'production_trace':False, 'statetune_samples_generated':0,
             'source_run':'author_validation/RESULTS.json',
             'split_policy':'fixed authored development projects; no confirmation/holdout split created',
             'role_protocol_applicability':'not a role prompt dataset; Agent runner owns shared role protocol construction',
             'similarity_audit':'cross-family similarity and dataset freeze owned by root integrator; not claimed completed here',
             'task_ids':list(CONTRACTS), 'seeds':{'RP-CLI-01':721809,'RP-CLI-02':561020,'RP-DATA-01':402918,'RP-DATA-02':331071},
             'agent_visibility':'only task.json.workspace_files and workspace_generators',
             'private_evaluation':'black-box child CLI processes on fixed-seed generated inputs, standalone standard library verifiers',
             'files':entries}
    (ROOT/'MANIFEST.json').write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')
    paths.append(ROOT/'MANIFEST.json')
    (ROOT/'SHA256SUMS').write_text(''.join(f"{hashlib.sha256(path.read_bytes()).hexdigest()}  {path.relative_to(ROOT)}\n" for path in sorted(paths)))


if __name__ == '__main__':
    if '--manifest-only' in sys.argv:
        manifest()
        raise SystemExit(0)
    build()
    successful = validate()
    manifest()
    raise SystemExit(0 if successful else 1)

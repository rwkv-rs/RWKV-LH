def public_url(value: str) -> str:
    parsed = urlsplit(value)
    hostname = parsed.hostname or ''
    port = f':{parsed.port}' if parsed.port else ''
    return urlunsplit((parsed.scheme, hostname + port, parsed.path, '', ''))


def public_runtime(settings: RuntimeSettings) -> dict:
    result = {field.name: getattr(settings, field.name) for field in fields(settings)
              if field.name not in PRIVATE_FIELDS}
    result['base_url'] = public_url(settings.base_url)
    for name in PRIVATE_FIELDS:
        result[name + '_configured'] = bool(getattr(settings, name))
    return result


def safe_error(exc: Exception) -> dict:
    # Exception bodies can echo URLs, request headers or credential values.
    names = sorted(set(re.findall(r'\b(?:RWKV_LH|RWKV|SUPERVISOR)_[A-Z][A-Z0-9_]*\b', str(exc))))
    return {'error_type': type(exc).__name__, 'configuration_fields_mentioned': names,
            'error_details_redacted': True}

def preflight(roles: dict, selector: NativeNetworkSelectorSettings,
              planner: SupervisorAPISettings | None) -> dict:
    """Only catalog/capability GETs; no tokenization or generation endpoints."""
    checks = {}
    executor = roles['executor']
    client = OpenAICompatibleRWKVClient(replace(
        executor, connect_timeout_seconds=5, read_timeout_seconds=15, retry_attempts=1,
    ))
    try:
        health = client.health()
        capabilities = client.capabilities()
        checks['rwkv_models'] = {
            'available': health.available,
            'configured_model_present': executor.model in health.models,
            'endpoint': public_url(executor.base_url) + '/models',
            'model': executor.model,
            'error_present': bool(health.error),
        }
        checks['rwkv_capabilities'] = {
            'durable_recurrent_state': capabilities.durable_recurrent_state,
            'recurrent_state_protocol': capabilities.recurrent_state_protocol,
            'expected_recurrent_state_protocol': NATIVE_STATE_PROTOCOL_VERSION,
            'native_state_supported': capabilities.durable_recurrent_state
                and capabilities.recurrent_state_protocol == NATIVE_STATE_PROTOCOL_VERSION,
        }
    except Exception as exc:
        checks['rwkv_preflight_error'] = safe_error(exc)
    finally:
        client.close()
    try:
        with requests.Session() as session:
            session.trust_env = False
            response = session.get(selector.base_url + '/healthz', timeout=(5, 15),
                                   allow_redirects=False)
            response.raise_for_status()
            value = response.json()
        expected = selector.runtime_identity()
        actual = value.get('runtime_identity', {})
        checks['selector'] = {
            'available': value.get('status') == 'ok',
            'identity_matches': actual == expected,
            'identity_match_by_field': {key: actual.get(key) == expected_value
                                        for key, expected_value in expected.items()},
            'endpoint': selector.base_url + '/healthz',
        }
    except Exception as exc:
        checks['selector'] = {'available': False, **safe_error(exc)}
    if planner is None:
        checks['strong_planner_stage_checker'] = {'available': False,
                                                  'reason': 'configuration_invalid'}
    else:
        supervisor = OpenAIGoalSupervisorClient(replace(
            planner, connect_timeout_seconds=5, read_timeout_seconds=15,
        ))
        try:
            health = supervisor.health()
            checks['strong_planner_stage_checker'] = {
                key: health.get(key) for key in (
                    'available', 'model_present', 'planner_model_present',
                    'stage_checker_model_present', 'latency_ms',
                )
            }
            checks['strong_planner_stage_checker']['error_present'] = bool(health.get('error'))
        finally:
            supervisor.close()
    checks['limits'] = [
        'GET health/catalog/capability checks do not prove actual generation correctness.',
        '13.3B model weight SHA and zero-profile absence require remote deployment attestation.',
        'No inference request, State bootstrap, benchmark, dataset or training performed.',
    ]
    return checks


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--mode', choices=('config', 'preflight'), default='config')
    parser.add_argument('--output', type=Path)
    args = parser.parse_args()
    if os.environ.get('WSL_DISTRO_NAME') != 'UbuntuRecovered':
        raise RuntimeError('this driver must run in WSL UbuntuRecovered')
    apply_current_zero_overrides()
    executor = RuntimeSettings.from_env()
    roles = {'executor': executor, **{
        role: RuntimeSettings.for_role(role, fallback=executor)
        for role in PROTOCOLS if role != 'executor'
    }}
    selector = NativeNetworkSelectorSettings.from_env()
    if selector is None:
        raise RuntimeError('current Selector configuration missing')
    planner = None
    try:
        # Match production: RWKV uses .env.local; Strong roles load allowed .env prefixes.
        planner = SupervisorAPISettings.from_env()
        planner_public = planner.public_dict()
        planner_public['base_url'] = public_url(planner.base_url)
        planner_public['sampling'] = 'not explicitly sent by current supervisor client; provider default'
        planner_public['policy'] = asdict(supervisor_policy_from_env(mode='static'))
    except Exception as exc:
        planner_public = {
            'configuration_valid': False, **safe_error(exc),
            'configuration_presence': {
                suffix: {
                    'canonical_configured': bool(os.environ.get(f'RWKV_LH_PLANNER_{suffix}')),
                    'legacy_configured': bool(os.environ.get(f'SUPERVISOR_{suffix}')),
                    'aliases_conflict': f'RWKV_LH_PLANNER_{suffix}' in os.environ
                        and f'SUPERVISOR_{suffix}' in os.environ
                        and os.environ[f'RWKV_LH_PLANNER_{suffix}'].strip()
                            != os.environ[f'SUPERVISOR_{suffix}'].strip(),
                } for suffix in ('BASE_URL', 'MODEL', 'API_KEY')
            },
            'stage_checker_model_configured': bool(os.environ.get('RWKV_LH_STAGE_CHECKER_MODEL')),
        }
    config = {
        'schema': 'rwkv-lh.current-zero-preflight-config.v1',
        'status': 'DRAFT_REQUIRES_FREEZE_REVIEW',
        'configuration_sources': ['.env.local', '.env (Strong-role allowed prefixes only)'],
        'architecture': STATEFUL_GOAL_LOOP_ARCHITECTURE,
        'server': 'rwkv-8222', 'server_ports': {'executor': 18234, 'selector': 18239},
        'roles': {role: {**public_runtime(settings),
                         'input_protocol': PROTOCOLS[role].INPUT_SCHEMA_VERSION}
                  for role, settings in roles.items()},
        'selector': {**selector.runtime_identity(), 'base_url': selector.base_url,
                     'connect_timeout_seconds': selector.connect_timeout_seconds,
                     'read_timeout_seconds': selector.read_timeout_seconds},
        'executor_profile_routing': os.environ[PROFILE_ROUTING_ENV],
        'strong_planner_stage_checker': planner_public,
        'semantic_sampling': {**LongHorizonModel._SAMPLING.to_dict(), 'semantic_resample_count': 0},
        'independent_run_argv': future_argv(),
        'execution_notes': [
            'This script has no benchmark execution mode; listed argv are review proposals only.',
            'Future argv must inherit apply_current_zero_overrides() from the reviewed process.',
            'Running the benchmark argv directly in a normal shell would reload unmodified .env.local.',
            'Two arm output directories differ; all other argv and captured settings match.',
            'Planner plan cache is disabled to prevent reuse of prior-run plans.',
            'No dataset or Holdout files are accessed by this driver.',
        ],
    }
    encoded = json.dumps(config, ensure_ascii=False, sort_keys=True, separators=(',', ':')).encode()
    report = {
        'created_at_utc': datetime.now(timezone.utc).isoformat(),
        'mode': args.mode, 'inference_requests': 0, 'benchmark_started': False,
        'PUBLIC_CONFIG': config,
        'PUBLIC_CONFIG_sha256': hashlib.sha256(encoded).hexdigest(),
        'driver_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'health': preflight(roles, selector, planner) if args.mode == 'preflight'
                  else {'status': 'not_requested_config_only'},
    }
    text = json.dumps(report, ensure_ascii=False, indent=2)
    if args.output:
        output = args.output.expanduser().resolve()
        if output.parent != ROUND or output.suffix != '.json':
            raise ValueError('output must be a JSON file directly under the current experiment')
        output.write_text(text + '\n', encoding='utf-8')
    print(text)
    return 0


if __name__ == '__main__':
    try:
        raise SystemExit(main())
    except Exception as failure:
        print(json.dumps({'status': 'configuration_error', **safe_error(failure)}))
        raise SystemExit(2) from None

"""Vary only JSON-mode field on an exact current Planner payload, for diagnosis."""
from pathlib import Path
from dataclasses import replace
from datetime import datetime,timezone
import argparse
import hashlib
import json
import sys
import time
ROOT=Path('/home/chase/GitHub/RWKV-LH')
sys.path.insert(0,str(ROOT))
import run_frozen_zero_baseline_r1_20260907 as driver
from rwkv_lh.supervisor_openai import SupervisorAPISettings,OpenAICompatibleSupervisorClient,_decode_supervisor_json_content
parser=argparse.ArgumentParser()
parser.add_argument('--variant', choices=['json','plain'],required=True)
parser.add_argument('--label',required=True)
args=parser.parse_args()
assert args.label.isalnum()
ROUND=ROOT/'data/experiments/ZERO_STATE_AGENT_BASELINE_R1_20260907'
source=ROUND/'PLANNER_WIRE_DIAGNOSTIC_R2/SUP-9bb02f7be600485aaedc.reconstructed_body.json'
body=json.loads(source.read_text())
if args.variant=='plain':del body['response_format']
output=ROUND/f'PLANNER_FULL_WIRE_{args.label}_{args.variant}.json'
assert not output.exists()
driver.apply_current_zero_overrides()
settings=replace(SupervisorAPISettings.from_env(),read_timeout_seconds=90.0)
client=OpenAICompatibleSupervisorClient(settings)
record={'at':datetime.now(timezone.utc).isoformat(),'purpose':'Owner-requested diagnosis, excluded from baseline/StateTune',
    'body_source':str(source.relative_to(ROOT)),'body_source_sha256':hashlib.sha256(source.read_bytes()).hexdigest(),
    'variant':args.variant,'request_body':body,
    'script_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    'request_body_sha256':hashlib.sha256(json.dumps(body,ensure_ascii=False,sort_keys=True).encode()).hexdigest(),
    'retry_attempts':1,'read_timeout_seconds':90.0,'production_sources_changed':False}
start=time.monotonic()
try:
    response=client._post_completion(settings.base_url+'/chat/completions',body)
    record['http_status']=response.status_code
    record['response_text']=response.text.replace(settings.api_key,'[REDACTED]')[:30000]
    record['response_sha256']=hashlib.sha256(response.content).hexdigest()
    if response.status_code==200:
        data=response.json()
        record['finish_reason']=data['choices'][0].get('finish_reason')
        record['usage']=data.get('usage')
        try:
            value,normalization=_decode_supervisor_json_content(data['choices'][0]['message']['content'])
            record['production_json_decoder_passed']=True
            record['decoded_value']=value
            record['normalization']=normalization
        except Exception as exc:
            record['production_json_decoder_passed']=False
            record['decode_error']=str(exc)
except Exception as exc:
    record['error_type']=type(exc).__name__
    record['error']=str(exc).replace(settings.api_key,'[REDACTED]')[:1200]
finally:
    record['elapsed_seconds']=round(time.monotonic()-start,3)
    client.close()
    output.write_text(json.dumps(record,ensure_ascii=False,indent=2)+'\n')
print(json.dumps({'output':str(output),'http_status':record.get('http_status'),
    'production_json_decoder_passed':record.get('production_json_decoder_passed'),'elapsed_seconds':record['elapsed_seconds']}))

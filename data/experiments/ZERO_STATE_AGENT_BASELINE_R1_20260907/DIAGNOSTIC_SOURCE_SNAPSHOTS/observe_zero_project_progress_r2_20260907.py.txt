"""Observe only this development run's exports and file stats; never open SQLite."""
from pathlib import Path
import argparse
import json
from datetime import datetime, timezone

ROOT = Path('/home/chase/GitHub/RWKV-LH')
ROUND = ROOT/'data/experiments/ZERO_STATE_AGENT_BASELINE_R1_20260907'
parser=argparse.ArgumentParser()
parser.add_argument('--arm', choices=['a','b'], required=True)
parser.add_argument('--compact', action='store_true')
args=parser.parse_args()
run=ROUND/f'real_project_zero_{args.arm}_r2'
frozen=json.loads((ROUND/'FROZEN_EXECUTION_MANIFEST_R2.json').read_text())
observed=[]
for ident in frozen['case_order']:
    case=run/'cases'/ident
    if not case.exists(): continue
    item={'task_id':ident,'audit_exported':False}
    audit=case/'audit.json'
    if audit.exists():
        try:
            doc=json.loads(audit.read_text())
        except json.JSONDecodeError:
            item['audit_export_in_progress']=True
        else:
            item.update(audit_exported=True, passed=doc.get('passed'),
                completed=doc.get('agent_completed'), failure=doc.get('failure'),
                status=(doc.get('run_state') or {}).get('status'),
                action_ledger_count=len(doc.get('action_ledger') or []),
                model_trace_count=len(doc.get('model_trace') or []))
    item['sqlite_component_stats']={p.name:{'bytes':p.stat().st_size,'mtime_utc':datetime.fromtimestamp(p.stat().st_mtime,timezone.utc).isoformat()}
        for p in sorted((case/'state').glob('long_horizon.db*'))}
    item['workspace_file_count']=sum(p.is_file() for p in (case/'workspace').rglob('*'))
    observed.append(item)
if args.compact:
    observed=[{k:v for k,v in item.items() if k not in ('sqlite_component_stats',)} |
        {'sqlite_component_bytes':sum(s['bytes'] for s in item['sqlite_component_stats'].values())}
        for item in observed]
print(json.dumps({'at':datetime.now(timezone.utc).isoformat(),'arm':args.arm,'frozen_denominator':12,
    'sqlite_opened':False,'observations':observed},ensure_ascii=False,indent=None if args.compact else 2))
